RadialMenu

Status: Experimental
Category: overlays
Import: @aeco-dev/gui
Emitted class(es): None by default

Radial menu anchored to a viewport element.

Each item is { icon, name, onSelect, enabled?, active? }. The application decides what a click does. Call setSelectionCount when the selection changes.

Source: src/components/menus/RadialMenu.js

Import

import { RadialMenu } from "@aeco-dev/gui";

Examples

Usage

const menu = new RadialMenu({
  viewport: host,
  items: [
    { icon: "content_copy", name: "Duplicate", onSelect: () => duplicate() },
  ],
});

RadialMenu

const wrap = new StackPanel({ isVertical: true }).gap("0.75rem");

  const status = new Caption("Last action: none");

  const viewport = new Container()
    .setStyle("position", ["relative"])
    .setStyle("height", ["340px"])
    .setStyle("border", ["1px dashed var(--dui-color-border, #444)"])
    .setStyle("borderRadius", ["8px"])
    .setStyle("overflow", ["hidden"]);

  const hint = new Caption("Hover this area and press P, or use the button below.");
  hint.setStyle("position", ["absolute"]);
  hint.setStyle("top", ["0.5rem"]);
  hint.setStyle("left", ["0.75rem"]);
  viewport.add(hint);

  const menu = new RadialMenu({
    viewport: viewport.dom,
    items: [
      { icon: "select_all", name: "Select all", onSelect: () => status.setTextContent("Last action: Select all") },
      { icon: "content_copy", name: "Duplicate", onSelect: () => status.setTextContent("Last action: Duplicate") },
      { icon: "category", name: "Group", onSelect: () => status.setTextContent("Last action: Group") },
      { icon: "visibility_off", name: "Hide", onSelect: () => status.setTextContent("Last action: Hide") },
      { icon: "delete", name: "Delete", onSelect: () => status.setTextContent("Last action: Delete") },
    ],
  });

  const toggleButton = new Button("Toggle pie menu")
    .addClass("primary")
    .onClick(() => menu.toggle());

  wrap.add(viewport);
  wrap.add(new StackPanel({ isVertical: false }).gap("0.75rem").setStyle("alignItems", ["center"]).add(toggleButton).add(status));
  return wrap;
